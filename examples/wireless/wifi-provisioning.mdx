title: "WiFi Provisioning"
---

```cpp
#include <iostream>
#include <string>

#include <ef/Device.hpp>

using namespace ef;

int main(int argc, char** argv) {
    if (argc < 3) {
        std::cerr << "usage: " << argv[0]
                  << " <ssid> <password> [country] [ble_mac]\n";
        return 2;
    }

    const std::string ssid = argv[1];
    const std::string password = argv[2];
    const std::string country = (argc >= 4) ? argv[3] : "";

    InitParameters init;
    if (argc >= 5) {
        init.input_type = INPUT_TYPE::STREAM;
        init.ble_address = argv[4];
    }

    Device device;
    ERROR_CODE status = device.open(init);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "open failed: " << to_string(status) << "\n";
        return 1;
    }

    status = device.wifi_add(ssid, password, country);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "wifi_add failed: " << to_string(status) << "\n";
        device.close();
        return 1;
    }

    status = device.wifi_select(ssid);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "wifi_select failed: " << to_string(status) << "\n";
        device.close();
        return 1;
    }

    const WirelessConfiguration& wifi =
        device.get_device_information().wireless;

    if (wifi.wifi_connected) {
        std::cout << "Connected to \"" << wifi.wifi_ssid
                  << "\" (ip " << wifi.wifi_ip_address << ").\n";
    } else {
        std::cout << "Provisioned \"" << ssid
                  << "\". Association is still in progress.\n";
    }

    device.close();
    return 0;
}
```
