```cpp
#include <iostream>
#include <string>

#include <ef/Device.hpp>

using namespace ef;

int main(int argc, char** argv) {
    const int target = (argc >= 2) ? std::stoi(argv[1]) : 150;

    InitParameters init;
    init.enable_imu = true;

    Device device;
    ERROR_CODE status = device.open(init);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "open failed: " << to_string(status) << "\n";
        return 1;
    }

    Mat frame;
    SensorsData imu;
    uint64_t imu_total = 0;
    int frames = 0;

    while (frames < target) {
        ERROR_CODE result = device.grab();
        if (result == ERROR_CODE::GRAB_TIMEOUT) {
            continue;
        }
        if (result != ERROR_CODE::SUCCESS) {
            std::cerr << "grab failed: " << to_string(result) << "\n";
            break;
        }

        device.retrieve_image(frame, VIEW::NV12);
        device.retrieve_imu(imu, TIME_REFERENCE::IMAGE);
        imu_total += imu.samples.size();
        ++frames;

        if (frames % 30 == 0) {
            std::cout << "frame " << frame.getFrameId()
                      << "  " << frame.getWidth() << "x" << frame.getHeight()
                      << "  imu +" << imu.samples.size()
                      << " (" << imu_total << " total)"
                      << "  motion=" << to_string(imu.motion_state) << "\n";
        }
    }

    std::cout << "captured " << frames << " frames, "
              << imu_total << " IMU samples\n";
    device.close();
    return 0;
}
```
