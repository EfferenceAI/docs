title: "Record and Download"
---

```cpp
#include <chrono>
#include <iostream>
#include <string>
#include <thread>

#include <ef/Device.hpp>

using namespace ef;

int main(int argc, char** argv) {
    const int seconds = (argc >= 2) ? std::stoi(argv[1]) : 3;

    Device device;
    ERROR_CODE status = device.open();
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "open failed: " << to_string(status) << "\n";
        return 1;
    }

    const std::string name = "example_recording";

    RecordingParameters recording;
    recording.target = RECORDING_TARGET::DEVICE_LOCAL;
    recording.name = name;

    status = device.enable_recording(recording);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "enable_recording failed: "
                  << to_string(status) << "\n";
        device.close();
        return 1;
    }

    std::this_thread::sleep_for(std::chrono::seconds(seconds));
    device.disable_recording();

    const std::string destination = name + ".mcap";
    status = device.download_recording(name, destination);
    if (status != ERROR_CODE::SUCCESS) {
        std::cerr << "download failed: " << to_string(status) << "\n";
        device.close();
        return 1;
    }

    std::cout << "saved " << destination << "\n";
    device.close();
    return 0;
}
```
