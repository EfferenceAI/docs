```cpp
#include <cstdio>
#include <ef/Device.hpp>

using namespace ef;

int main() {
    Device dev;
    if (dev.open() != ERROR_CODE::SUCCESS) return 1;

    SensorsData sensors;
    uint64_t total = 0;

    while (total < 2000) {
        ERROR_CODE ec = dev.grab();
        if (ec == ERROR_CODE::GRAB_TIMEOUT) continue;
        if (ec != ERROR_CODE::SUCCESS && ec != ERROR_CODE::CORRUPTED_FRAME) break;

        if (dev.retrieve_imu(sensors) != ERROR_CODE::SUCCESS) continue;
        total += sensors.samples.size();

        if (!sensors.samples.empty()) {
            const ImuSample& s = sensors.samples.back();   // newest
            std::printf("accel [%+6.2f %+6.2f %+6.2f] m/s²  "
                        "gyro [%+6.3f %+6.3f %+6.3f] rad/s  %s\n",
                        s.acceleration[0], s.acceleration[1], s.acceleration[2],
                        s.angular_velocity[0], s.angular_velocity[1],
                        s.angular_velocity[2],
                        to_string(sensors.motion_state));
        }
    }

    dev.close();
    return 0;
}
```

## Walkthrough

- `retrieve_imu()` drains **every sample since the previous call** — at
  ~400 Hz IMU and 30 fps video that is roughly 13 samples per grab. Iterate
  `sensors.samples` for all of them; use `.back()` for a live display.
- `sensors.dropped` counts ring overruns. Non-zero means your loop stalled
  long enough for the host buffer to wrap — a slow consumer, not wire loss.
- A stationary device should read `|accel| ≈ 9.81 m/s²`, gyro near zero
  (the residual is the gyro bias), and `MOTION_STATE::STATIC`.
- Sample timestamps share the device clock with `Mat::timestamp`, so pairing
  IMU windows to frames is a direct comparison — no offset calibration.

Video-free variant: set `InitParameters::enable_imu = true` and simply never
call `retrieve_image()`; frames are superseded in the SDK with no cost.
